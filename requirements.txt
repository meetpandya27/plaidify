# Runtime dependencies: the direct pins, reviewed by hand.
#
# The image and CI install the hash-locked requirements*.lock files, which pin
# every transitive dependency with its hashes. After changing a pin here or in
# requirements-dev.txt / requirements-kms.txt, regenerate all three locks with
# pip-tools under Python 3.11 (the image's Python), then check them:
#
#   pip-compile --generate-hashes --allow-unsafe --strip-extras \
#       --output-file=requirements.lock requirements.txt
#   pip-compile --generate-hashes --allow-unsafe --strip-extras \
#       --output-file=requirements-dev.lock requirements-dev.txt
#   pip-compile --generate-hashes --allow-unsafe --strip-extras \
#       --constraint=requirements.lock \
#       --output-file=requirements-kms.lock requirements-kms.txt
#   python scripts/check_requirements_locks.py
